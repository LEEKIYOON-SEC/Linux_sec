# webshell_regex.txt — 17_webshell 모듈이 grep -E -f 로 읽는 정규식 모음
#
# 각 줄은 grep -E (POSIX ERE) 호환 정규식 한 개.
# '#' 으로 시작하는 줄과 빈 줄은 grep -f 가 자체 처리하지 않으므로 패턴이 됨에 주의.
# (실제로는 빈 패턴이 모든 줄 매치 → 노이즈) 그래서 grep -v '^#' 로 사전 필터링을
# 17_webshell.sh 가 처리한다.
#
# 작성 원칙:
#   - false positive 최소화. 정상 프레임워크 코드에서 흔히 나오는 표현은 제외.
#   - "공격 의도가 명확한" 조합만 (예: eval(base64_decode), system($_GET)).
#   - 운영 후 노이즈가 잡히면 patterns/custom_iocs/whitelist.txt 가 아니라
#     이 파일을 좁혀 잡거나, secchk.conf 의 점검 대상에서 제외.
#
# 참고: grep -E -f 는 한 줄을 하나의 ERE 로 읽는다. 줄 사이는 OR.

# --- PHP: 난독화된 코드 실행 (공격 패턴) ---
# eval(base64_decode(...))  형태가 가장 흔하다.
eval[[:space:]]*\([[:space:]]*(base64_decode|gzinflate|gzuncompress|gzdecode|str_rot13|convert_uudecode|hex2bin)

# --- PHP: 사용자 입력을 그대로 명령 실행 ---
(system|exec|passthru|shell_exec|popen|proc_open|pcntl_exec)[[:space:]]*\([[:space:]]*\$_(GET|POST|REQUEST|COOKIE|SERVER)

# --- PHP: assert 사용한 코드 실행 (PHP 7.2+ 에서 사실상 폐기, 정상 코드에는 거의 없음) ---
assert[[:space:]]*\([[:space:]]*\$_(GET|POST|REQUEST|COOKIE)

# --- PHP: preg_replace /e 플래그 (PHP 5.5+ deprecated, 5.6+ E_DEPRECATED, 7+ 제거) ---
# 정상 코드에 남아있을 가능성이 매우 낮음.
preg_replace[[:space:]]*\([^,]*['"][^'"]*/e['"]

# --- PHP: create_function 동적 코드 (PHP 7.2 deprecated) ---
create_function[[:space:]]*\([[:space:]]*['"][^'"]*\$_

# --- PHP: 짧은 한 줄 백도어 (<?=$_GET[0](...)?>) 패턴 ---
\<\?=[[:space:]]*\$_(GET|POST|REQUEST)

# --- JSP: Runtime.exec / ProcessBuilder (정상 코드에도 있을 수 있으나 웹 페이지에는 드뭄) ---
Runtime\.getRuntime\(\)\.exec[[:space:]]*\(
new[[:space:]]+ProcessBuilder[[:space:]]*\(

# --- JSP: 동적 클래스 로딩 백도어 패턴 ---
ClassLoader\.getSystemClassLoader\(\)\.loadClass

# --- ASP: WScript.Shell 객체 (정상 ASP 에서도 쓰이지만, 일반 운영 서버 코드에선 드뭄) ---
Server\.CreateObject[[:space:]]*\([[:space:]]*["'](WScript\.Shell|Shell\.Application|Scripting\.FileSystemObject)

# --- ASPX/ASP: 사용자 입력을 곧장 Process.Start 등에 넘기는 패턴 ---
Process\.Start[[:space:]]*\([[:space:]]*Request\.

# --- 공통: 거의 모든 웹쉘에 들어가는 매직 헤더 (c99/r57 변종) ---
[Cc]99[Ss]hell|[Rr]57[Ss]hell|FilesMan|FilesTools|wso[[:space:]]*shell
